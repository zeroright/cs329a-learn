"""自编: unbiased pass@k estimator + coverage curve (Monkeys / HumanEval style)."""
from __future__ import annotations
import math
from collections import Counter

def unbiased_pass_at_k(n: int, c: int, k: int) -> float:
    """HumanEval unbiased estimator: 1 - C(n-c, k) / C(n, k)."""
    if n - c < k:
        return 1.0
    return 1.0 - math.comb(n - c, k) / math.comb(n, k)

def coverage_power_law(ks, A=0.55, B=0.35):
    """Toy C(K) ≈ 1 - A * K^{-B} as in Large Language Monkeys."""
    return [1.0 - A * (k ** (-B)) for k in ks]

def demo():
    # Fake per-problem: n samples, c correct
    problems = [(16, 3), (16, 0), (16, 8), (16, 1), (8, 2)]
    for k in (1, 2, 4, 8):
        scores = [unbiased_pass_at_k(n, c, min(k, n)) for n, c in problems]
        print(f"pass@{k} mean={sum(scores)/len(scores):.3f}  per={['%.2f'%s for s in scores]}")
    ks = [1, 2, 4, 8, 16, 32, 64]
    cov = coverage_power_law(ks)
    print("coverage toy:", list(zip(ks, [round(x, 3) for x in cov])))
    # majority vs oracle best-of-n
    samples = ["42", "7", "42", "42", "99", "7", "42", "3"]
    truth = "42"
    maj = Counter(samples).most_common(1)[0][0]
    best = truth if truth in samples else samples[0]
    print(f"majority={maj} best-of-n(oracle)={best} maj_ok={maj==truth}")

if __name__ == "__main__":
    demo()

"""自编: minimal ReAct loop with mock LLM + tools (no paid API)."""
from __future__ import annotations

TOOLS = {
    "search": lambda q: f"[search:{q}] stub hit: Stanford CS329A teaches self-improving agents.",
    "calc": lambda expr: str(eval(expr, {"__builtins__": {}}, {})),  # toy only
}

def mock_llm(step: int, question: str, obs: str | None) -> tuple[str, str | None, str | None]:
    """Return (thought, action_or_none, final_or_none)."""
    if step == 0:
        return ("Need course topic facts.", "search[CS329A overview]", None)
    if step == 1:
        return ("Have overview; compute 2+2 as sanity.", "calc[2+2]", None)
    return ("Enough evidence.", None, f"Answer: agents + verifiers. Obs calc={obs}")

def run_react(question: str, max_steps: int = 5) -> str:
    transcript = []
    obs = None
    for t in range(max_steps):
        thought, action, final = mock_llm(t, question, obs)
        transcript.append(f"Thought: {thought}")
        if final:
            transcript.append(f"Final: {final}")
            return "\n".join(transcript)
        assert action and "[" in action
        name, arg = action[:-1].split("[", 1)
        obs = TOOLS[name](arg)
        transcript.append(f"Action: {action}\nObservation: {obs}")
    return "\n".join(transcript) + "\nFinal: (max steps)"

if __name__ == "__main__":
    print(run_react("What is CS329A about?"))
